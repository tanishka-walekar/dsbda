import pandas as pd
import numpy as np
from sklearn.preprocessing import MinMaxScaler
# 1. Create academic dataset
df = pd.DataFrame({
    'Name': ['A','B','C','D','E'],
    'Marks': [85, 90, np.nan, 120, 70],  
    'Study_Hours': [5, 6, 7, np.nan, 4]
})
df
df.info()
df.describe()
# 2. Handle missing values
df['Marks'] = df['Marks'].fillna (df['Marks'].mean())
df['Study_Hours'] = df['Study_Hours'].fillna(df['Study_Hours'].mean())
df
# Fix inconsistency (Marks > 100)
df.loc[df['Marks'] > 100, 'Marks'] = 100
df
# 3. Detect outliers using IQR
Q1 = df['Marks'].quantile(0.25)
Q3 = df['Marks'].quantile(0.75)
IQR = Q3 - Q1
outliers = df[(df['Marks'] < (Q1 - 1.5*IQR)) | (df['Marks'] > (Q3 + 1.5*IQR))]
print("Outliers:\n", outliers)
df
# 4. Data Transformation

# Log transformation
df['Marks_log'] = np.log(df['Marks'])
df
# Normalization (0 to 1)
scaler = MinMaxScaler()
df[['Marks_norm']] = scaler.fit_transform(df[['Marks']])
df
